```python
import numpy as np
import soundfile as sf
from scipy import signal

def preprocess_audio(audio_path, sample_rate=16000):
    """Load and preprocess audio for model inference"""
    audio, sr = sf.read(audio_path)
    
    if audio.ndim == 2:
        audio = np.mean(audio, axis=1)
    
    if sr != sample_rate:
        audio = signal.resample(audio, int(len(audio) * sample_rate / sr))
    
    return audio.astype(np.float32)
```

```python
from huggingface_hub import hf_hub_download
import os

def load_model(model_name, model_dir):
    """Load GGUF model from HuggingFace"""
    repo_id = f"mudler/{model_name}"
    local_dir = os.path.join(model_dir, repo_id)
    
    # Download model files
    files = ["model.gguf", "config.json"]  # Adjust based on actual model files
    for filename in files:
        hf_hub_download(repo_id=repo_id, filename=filename, local_dir=local_dir)
    
    return local_dir
```

**Step 2: Write model inference function**

```python
import onnxruntime as ort

def run_inference(model_path, audio):
    """Run ONNX model inference"""
    session = ort.InferenceSession(model_path)
    inputs = session.get_inputs()
    
    # Prepare input
    input_name = inputs[0].name
    input_data = {input_name: audio.reshape(1, -1)}
    
    # Run inference
    outputs = session.run(None, input_data)
    return outputs
```

```python
import matplotlib.pyplot as plt

def create_age_gender_plot(result):
    fig, ax = plt.subplots(figsize=(8, 4))
    ax.bar(['Age', 'Gender'], [result['confidence'], result['confidence']])
    ax.set_title(f"Age/Gender Detection - {result['age']}, {result['gender']}")
    ax.set_ylabel('Confidence')
    ax.set_ylim([0, 1])
    plt.tight_layout()
    return fig

def create_emotion_plot(result):
    fig, ax = plt.subplots(figsize=(8, 4))
    emotions = list(result['emotions'].keys())
    values = list(result['emotions'].values())
    
    ax.pie(values, labels=emotions, autopct='%1.1f%%')
    ax.set_title('Emotion Detection')
    plt.tight_layout()
    return fig

def create_speaker_plot(result):
    fig, ax = plt.subplots(figsize=(8, 4))
    ax.bar(['Similarity', 'Confidence'], [result['similarity'], result['confidence']])
    ax.set_title('Speaker Verification')
    ax.set_ylabel('Score')
    ax.set_ylim([0, 1])
    plt.tight_layout()
    return fig
```

**Step 2: Create multi-model layout**

```python
def create_multi_plot(results):
    fig, axes = plt.subplots(len(results), 1, figsize=(10, 4 * len(results)))
    
    for idx, (model_name, result) in enumerate(results.items()):
        if model_name == "age_gender":
            axes[idx] = create_age_gender_plot(result)
        elif model_name == "emotion":
            axes[idx] = create_emotion_plot(result)
        elif model_name == "speaker_verification":
            axes[idx] = create_speaker_plot(result)
    
    plt.tight_layout()
    return fig
```
